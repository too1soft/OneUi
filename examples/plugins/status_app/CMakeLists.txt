project(StatusPluginConsumer LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(OneUI CONFIG REQUIRED)

oneui_target_components(status_app REGISTRAR register_components COMPONENTS OneUI::status)
oneui_target_view(status_app NAME StatusDemo SOURCE Status.one VM_HEADER vm.h)
