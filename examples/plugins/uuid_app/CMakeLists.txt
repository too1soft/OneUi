cmake_minimum_required(VERSION 3.24)
project(UuidPluginConsumer LANGUAGES C CXX)
if(DEFINED ONEUI_PROJECT_PLUGINS)
    include("${ONEUI_PROJECT_PLUGINS}")
else()
    find_package(OneUIUuid CONFIG REQUIRED)
endif()
add_executable(uuid_plugin_consumer main.c)
set_target_properties(uuid_plugin_consumer PROPERTIES LINKER_LANGUAGE CXX)
oneui_target_plugins(uuid_plugin_consumer REGISTRAR register_uuid_plugins PLUGINS OneUI::uuid)
add_executable(uuid_plugin_tests tests.cpp)
target_compile_features(uuid_plugin_tests PRIVATE cxx_std_17)
oneui_target_plugins(uuid_plugin_tests REGISTRAR register_uuid_test_plugins PLUGINS OneUI::uuid)
enable_testing()
add_test(NAME uuid_c_consumer COMMAND uuid_plugin_consumer)
add_test(NAME uuid_contract COMMAND uuid_plugin_tests)
