  target_link_libraries(oneui-lab-process-tests PRIVATE shell32)

  find_package(Threads REQUIRED)

if(UNIX)
  set(ONEUI_EXISTING_BUILD_DIR "" CACHE PATH "Matching existing OneUI SDK output")
  if(APPLE)
    enable_language(OBJCXX)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0")
    set(lab_library "${ONEUI_EXISTING_BUILD_DIR}/liboneui.dylib")
    set(lab_platform platform_macos.mm)
  else()
    set(lab_library "${ONEUI_EXISTING_BUILD_DIR}/liboneui.so")
    set(lab_platform platform_linux.cpp)
  endif()
  if(NOT EXISTS "${lab_library}")
    message(FATAL_ERROR "Set ONEUI_EXISTING_BUILD_DIR to a matching built SDK output")
  endif()
  add_library(oneui SHARED IMPORTED)
  set_target_properties(oneui PROPERTIES IMPORTED_LOCATION "${lab_library}" INTERFACE_INCLUDE_DIRECTORIES "${ONEUI_SOURCE_ROOT}/include")
  include("${ONEUI_SOURCE_ROOT}/cmake/OneUIView.cmake")
  if(APPLE)
    add_executable(oneui-performance-lab MACOSX_BUNDLE ${lab_sources} ${lab_platform})
    target_link_libraries(oneui-performance-lab PRIVATE "-framework AppKit")
    target_compile_options(oneui-performance-lab PRIVATE $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>)
    set_target_properties(oneui-performance-lab PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "org.oneui.performance-lab" MACOSX_BUNDLE_BUNDLE_NAME "OneUI 试验台")
  else()
    add_executable(oneui-performance-lab ${lab_sources} ${lab_platform})
  endif()
  target_link_libraries(oneui-performance-lab PRIVATE oneui)
  target_include_directories(oneui-performance-lab PRIVATE "${ONEUI_SOURCE_ROOT}/src")
  option(ONEUI_LAB_MESH_AVAILABLE "SDK includes the private circle mesh experiment" OFF)
  if(ONEUI_LAB_MESH_AVAILABLE)
    target_compile_definitions(oneui-performance-lab PRIVATE ONEUI_LAB_MESH_AVAILABLE=1)
  endif()
  foreach(view Connections Gallery Recipes)
    oneui_target_view(oneui-performance-lab NAME ${view} SOURCE views/${view}.one)
  endforeach()
  include(rust_owner.cmake)
  if(APPLE)
    include(macos_bundle.cmake)
  endif()
  include(compiled_examples.cmake)
  include(workspace_tests.cmake)
  return()
endif()
