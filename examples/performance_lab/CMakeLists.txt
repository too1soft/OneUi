cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(oneui_performance_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ONEUI_BUILD_STATIC OFF CACHE BOOL "" FORCE)
set(ONEUI_ENABLE_TEST_FRAME_CAPTURE ON CACHE BOOL "" FORCE)
set(ONEUI_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "OneUI source with declarative APIs")
set(ONEUI_ENABLE_YOGA ON CACHE BOOL "" FORCE)
add_subdirectory("${ONEUI_SOURCE_ROOT}" sdk EXCLUDE_FROM_ALL)
# The private SkMesh experiment is not compiled into normal SDK builds.
target_compile_definitions(oneui_platform_win32 PRIVATE ONEUI_CIRCLE_MESH_EXPERIMENT=1 ONEUI_LAB_RENDER_TESTS=1)
include("${ONEUI_SOURCE_ROOT}/cmake/OneUIView.cmake")
add_executable(oneui-performance-lab WIN32 main.cpp)
target_link_libraries(oneui-performance-lab PRIVATE oneui psapi)
target_compile_options(oneui-performance-lab PRIVATE /utf-8 /W4)
target_compile_definitions(oneui-performance-lab PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_custom_command(TARGET oneui-performance-lab POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:oneui> $<TARGET_FILE_DIR:oneui-performance-lab>)
add_executable(oneui-editor-tests editor_tests.cpp)
target_link_libraries(oneui-editor-tests PRIVATE oneui)
target_compile_options(oneui-editor-tests PRIVATE /utf-8)
target_compile_definitions(oneui-editor-tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(oneui-editor-tests PRIVATE "${ONEUI_SOURCE_ROOT}/tests")
add_executable(oneui-connections-tests connections_tests.cpp)
target_link_libraries(oneui-connections-tests PRIVATE oneui)
target_compile_options(oneui-connections-tests PRIVATE /utf-8)
target_compile_definitions(oneui-connections-tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(oneui-connections-tests PRIVATE "${ONEUI_SOURCE_ROOT}/tests")
oneui_target_view(oneui-performance-lab NAME Connections SOURCE views/Connections.one)
oneui_target_view(oneui-connections-tests NAME Connections SOURCE views/Connections.one)
add_executable(oneui-authoring-tests authoring_tests.cpp)
target_link_libraries(oneui-authoring-tests PRIVATE oneui)
target_compile_options(oneui-authoring-tests PRIVATE /utf-8)
target_compile_definitions(oneui-authoring-tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(oneui-authoring-tests PRIVATE "${ONEUI_SOURCE_ROOT}/tests")
oneui_target_view(oneui-authoring-tests NAME Connections SOURCE views/Connections.one)
foreach(target oneui-performance-lab oneui-connections-tests oneui-authoring-tests)
  oneui_target_view(${target} NAME Gallery SOURCE views/Gallery.one)
endforeach()
add_executable(oneui-gallery-tests gallery_tests.cpp)
target_link_libraries(oneui-gallery-tests PRIVATE oneui)
target_compile_options(oneui-gallery-tests PRIVATE /utf-8)
target_compile_definitions(oneui-gallery-tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(oneui-gallery-tests PRIVATE "${ONEUI_SOURCE_ROOT}/tests")
oneui_target_view(oneui-gallery-tests NAME Gallery SOURCE views/Gallery.one)
add_executable(oneui-renderer-tests renderer_tests.cpp)
target_link_libraries(oneui-renderer-tests PRIVATE oneui)
target_compile_options(oneui-renderer-tests PRIVATE /utf-8)
target_compile_definitions(oneui-renderer-tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_executable(oneui-particle-tests particle_tests.cpp)
target_link_libraries(oneui-particle-tests PRIVATE oneui opengl32)
target_compile_options(oneui-particle-tests PRIVATE /utf-8)
target_compile_definitions(oneui-particle-tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(oneui-particle-tests PRIVATE "${ONEUI_SOURCE_ROOT}/tests")
foreach(target oneui-performance-lab oneui-particle-tests)
  target_include_directories(${target} PRIVATE "${ONEUI_SOURCE_ROOT}/src")
endforeach()

add_executable(oneui-lifecycle-tests lifecycle_tests.cpp)
target_link_libraries(oneui-lifecycle-tests PRIVATE oneui psapi)
target_compile_options(oneui-lifecycle-tests PRIVATE /utf-8)
target_compile_definitions(oneui-lifecycle-tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(oneui-lifecycle-tests PRIVATE "${ONEUI_SOURCE_ROOT}/src")
