cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(oneui_declarative_demo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(ONEUI_DEMO_SDK_TESTS "Also configure SDK regression tests" OFF)
set(BUILD_TESTING ${ONEUI_DEMO_SDK_TESTS} CACHE BOOL "" FORCE)
include(CTest)
enable_testing()
set(ONEUI_BUILD_STATIC OFF CACHE BOOL "" FORCE)
set(ONEUI_ENABLE_YOGA ON CACHE BOOL "" FORCE)
set(ONEUI_ENABLE_TEST_FRAME_CAPTURE ON CACHE BOOL "" FORCE)
set(ONEUI_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "OneUI source")
add_subdirectory("${ONEUI_SOURCE_ROOT}" sdk EXCLUDE_FROM_ALL)
include("${ONEUI_SOURCE_ROOT}/cmake/OneUIView.cmake")
add_executable(oneui-hello hello.cpp)
target_link_libraries(oneui-hello PRIVATE oneui)
target_compile_options(oneui-hello PRIVATE /utf-8)
add_executable(oneui-details details.cpp)
target_link_libraries(oneui-details PRIVATE oneui)
target_compile_options(oneui-details PRIVATE /utf-8)
add_executable(oneui-declarative-demo main.cpp)
target_link_libraries(oneui-declarative-demo PRIVATE oneui psapi)
target_compile_options(oneui-declarative-demo PRIVATE /utf-8 /W3 /wd4251)
target_compile_definitions(oneui-declarative-demo PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN DEMO_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
oneui_target_view(oneui-declarative-demo NAME Demo SOURCE views/Demo.one VM_HEADER vm.h)
add_executable(oneui-declarative-tests tests.cpp)
target_link_libraries(oneui-declarative-tests PRIVATE oneui)
target_include_directories(oneui-declarative-tests PRIVATE "${ONEUI_SOURCE_ROOT}/tests")
target_compile_options(oneui-declarative-tests PRIVATE /utf-8)
target_compile_definitions(oneui-declarative-tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
oneui_target_view(oneui-declarative-tests NAME Demo SOURCE views/Demo.one VM_HEADER vm.h)
add_test(NAME declarative_runtime COMMAND oneui-declarative-tests)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME declarative_compiler COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_compiler.py" "$<TARGET_FILE:oneui-viewc>" "${CMAKE_CXX_COMPILER}" "${ONEUI_SOURCE_ROOT}/include")
