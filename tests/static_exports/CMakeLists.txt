cmake_minimum_required(VERSION 3.16)
project(oneui_static_export_contract LANGUAGES CXX)
get_filename_component(oneui_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${oneui_root}/cmake/OneUIStatic.cmake")
add_library(export_objects OBJECT exported.cpp)
target_include_directories(export_objects PUBLIC "${oneui_root}/include")
target_compile_definitions(export_objects PRIVATE ONEUI_BUILDING_DLL)
oneui_static_objects(static_objects export_objects)
add_library(export_contract SHARED $<TARGET_OBJECTS:export_objects>)
add_library(static_contract STATIC $<TARGET_OBJECTS:static_objects>)
target_include_directories(static_contract PUBLIC "${oneui_root}/include")
target_compile_definitions(static_contract PUBLIC ONEUI_STATIC)
oneui_static_link_options(static_contract)
add_executable(static_consumer main.cpp)
target_link_libraries(static_consumer PRIVATE static_contract)
