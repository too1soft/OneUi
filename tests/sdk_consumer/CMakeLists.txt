cmake_minimum_required(VERSION 3.16)
project(oneui_sdk_consumers LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(ONEUI_SOURCE_DIR)
    # Separate development/static product, not part of the dynamic SDK.
    add_subdirectory("${ONEUI_SOURCE_DIR}" oneui-source)
    set(consumer_library OneUI::oneui_static)
else()
    find_package(OneUI CONFIG REQUIRED)
    set(consumer_library OneUI::oneui)
endif()
add_executable(oneui_c_consumer main.c)
add_executable(oneui_cpp_consumer main.cpp)
target_link_libraries(oneui_c_consumer PRIVATE ${consumer_library})
target_link_libraries(oneui_cpp_consumer PRIVATE ${consumer_library})
if(ONEUI_SOURCE_DIR)
    set_target_properties(oneui_c_consumer PROPERTIES LINKER_LANGUAGE CXX)
endif()
enable_testing()
add_test(NAME oneui_c_consumer COMMAND oneui_c_consumer)
add_test(NAME oneui_cpp_consumer COMMAND oneui_cpp_consumer)
set_tests_properties(oneui_c_consumer oneui_cpp_consumer PROPERTIES TIMEOUT 15)
