cmake_minimum_required(VERSION 3.24)
project(OneUIStatus VERSION 0.1.0 LANGUAGES CXX)
if(NOT TARGET OneUI::oneui)
    find_package(OneUI CONFIG REQUIRED)
endif()
include("${OneUI_DIR}/OneUIComponents.cmake")
if(NOT ONEUI_SDK_ABI STREQUAL "46")
    message(FATAL_ERROR "OneUI Status 0.1 requires OneUI SDK ABI 46")
endif()
add_library(oneui_status STATIC src/component.cpp)
add_library(OneUI::status ALIAS oneui_status)
target_compile_features(oneui_status PUBLIC cxx_std_17)
target_link_libraries(oneui_status PUBLIC OneUI::oneui)
target_include_directories(oneui_status PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
oneui_component_contract(oneui_status SOURCE Status.onecomponents NAME status_contract)
set_target_properties(oneui_status PROPERTIES EXPORT_NAME status POSITION_INDEPENDENT_CODE ON
    ONEUI_COMPONENT_ID oneui.status ONEUI_COMPONENT_ENTRY oneui_status_component_v1
    ONEUI_COMPONENT_HEADER oneui/plugins/status.h ONEUI_COMPONENT_CONTRACT "${CMAKE_CURRENT_SOURCE_DIR}/Status.onecomponents")
execute_process(COMMAND git -C "${CMAKE_CURRENT_SOURCE_DIR}" rev-parse HEAD OUTPUT_VARIABLE _commit OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
execute_process(COMMAND git -C "${CMAKE_CURRENT_SOURCE_DIR}" status --porcelain OUTPUT_VARIABLE _dirty ERROR_QUIET)
if(_dirty)
    set(_dirty_json true)
else()
    set(_dirty_json false)
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/oneui-plugin-build-$<CONFIG>.json" CONTENT
    "{\"schemaVersion\":1,\"sourceCommit\":\"${_commit}\",\"sourceDirty\":${_dirty_json},\"configuration\":\"$<CONFIG>\",\"system\":\"${CMAKE_SYSTEM_NAME}\",\"processor\":\"${CMAKE_SYSTEM_PROCESSOR}\",\"pointerBytes\":${CMAKE_SIZEOF_VOID_P},\"compiler\":\"${CMAKE_CXX_COMPILER_ID}\",\"compilerVersion\":\"${CMAKE_CXX_COMPILER_VERSION}\",\"runtime\":\"${CMAKE_MSVC_RUNTIME_LIBRARY}\",\"linkage\":\"SHARED\",\"coreABI\":46}\n")
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS oneui_status EXPORT OneUIStatusTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES Status.onecomponents oneui-plugin.json LICENSE DESTINATION share/oneui/plugins/oneui.status)
install(EXPORT OneUIStatusTargets NAMESPACE OneUI:: DESTINATION lib/cmake/OneUIStatus)
install(FILES cmake/OneUIStatusConfig.cmake DESTINATION lib/cmake/OneUIStatus)
